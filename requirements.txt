-- 3. Add generated dedup_key column for composite deduplication.
do $$ begin
  if not exists (select 1 from information_schema.columns
    where table_name='job_results' and column_name='dedup_key') then
    alter table job_results add column dedup_key text
      generated always as (
        lower(trim(company_name)) || '|' ||
        lower(trim(title))        || '|' ||
        lower(trim(coalesce(location, '')))
      ) stored;
  end if;
end $$;

create unique index if not exists idx_job_results_dedup_key
  on job_results(dedup_key);
